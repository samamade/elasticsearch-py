from flask import Flask, request, render_template
from elasticsearch import Elasticsearch

app = Flask(__name__)

# الاتصال بـ Elasticsearch
es = Elasticsearch("http://localhost:9200")

# إنشاء فهرس جديد (إذا لم يكن موجودًا)
def create_index():
    if not es.indices.exists(index="sama1_index"):
        es.indices.create(index="sama1_index", ignore=400)

create_index()

# الصفحة الرئيسية لواجهة البحث
@app.route('/')
def home():
    return render_template('index.html')

# معالج الاستعلامات
@app.route('/search', methods=['GET'])
def search():
    query = request.args.get('query')
    
    # استعلام Elasticsearch
    if query:
        res = es.search(index="sama1_index", body={
            "query": {
                "match": {
                    "content": query
                }
            }
        })
        results = res['hits']['hits']
    else:
        results = []
    
    return render_template('index.html', query=query, results=results)

if __name__ == '__main__':
    app.run(debug=True)
